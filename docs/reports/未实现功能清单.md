#### 12.1 空值/Null值生成器

```python
# dataforge/generators/special/empty_value.py

from typing import Any, Optional, Union
from dataforge.core.generator import DataGenerator, GeneratorConfig

class EmptyValueGenerator(DataGenerator[Optional[Any]]):
    """空值/Null值生成器"""

    def __init__(self, config: GeneratorConfig):
        super().__init__(config)
        self.empty_type = config.parameters.get("type", "null")  # null, empty_string, empty_array, empty_object
        self.frequency = config.parameters.get("frequency", 0.1)

    def _generate_raw(self, context=None) -> Optional[Any]:
        import random
        if random.random() > self.frequency:
            return None  # 不生成空值，返回None让调用者决定

        if self.empty_type == "null":
            return None
        elif self.empty_type == "empty_string":
            return ""
        elif self.empty_type == "empty_array":
            return []
        elif self.empty_type == "empty_object":
            return {}
        elif self.empty_type == "undefined":
            return "undefined"  # JavaScript风格
        else:
            return None
```

```python
from typing import Any, Union
from dataforge.core.generator import DataGenerator, GeneratorConfig
import sys

class BoundaryValueGenerator(DataGenerator[Union[int, float, str]]):
    """边界值/极端值生成器"""

    def __init__(self, config: GeneratorConfig):
        super().__init__(config)
        self.data_type = config.parameters.get("type", "numeric")  # numeric, string, datetime
        self.boundary_type = config.parameters.get("boundary_type", "min")  # min, max, near_min, near_max, zero

    def _generate_raw(self, context=None) -> Union[int, float, str]:
        if self.data_type == "numeric":
            return self._generate_numeric_boundary()
        elif self.data_type == "string":
            return self._generate_string_boundary()
        elif self.data_type == "datetime":
            return self._generate_datetime_boundary()

    def _generate_numeric_boundary(self) -> Union[int, float]:
        if self.boundary_type == "min":
            return -sys.maxsize - 1
        elif self.boundary_type == "max":
            return sys.maxsize
        elif self.boundary_type == "zero":
            return 0
        elif self.boundary_type == "near_min":
            return -sys.maxsize
        elif self.boundary_type == "near_max":
            return sys.maxsize - 1
```

### 2. CSV/TSV生成器实现方案

```python
# dataforge/generators/structured/csv_generator.py

from typing import Any, List, Dict
from dataforge.core.generator import DataGenerator, GeneratorConfig
import csv
import io

class CSVGenerator(DataGenerator[str]):
    """CSV/TSV生成器"""

    def __init__(self, config: GeneratorConfig):
        super().__init__(config)
        self.delimiter = config.parameters.get("delimiter", ",")
        self.include_header = config.parameters.get("include_header", True)
        self.columns = config.parameters.get("columns", [])  # [{"name": "id", "type": "integer"}, ...]
        self.row_count = config.parameters.get("row_count", 10)

    def _generate_raw(self, context=None) -> str:
        output = io.StringIO()
        writer = csv.writer(output, delimiter=self.delimiter)

        # 写入标题行
        if self.include_header:
            headers = [col["name"] for col in self.columns]
            writer.writerow(headers)

        # 写入数据行
        for _ in range(self.row_count):
            row = []
            for col in self.columns:
                # 根据列类型生成数据
                value = self._generate_column_value(col, context)
                row.append(value)
            writer.writerow(row)

        return output.getvalue()

    def _generate_column_value(self, column: Dict[str, Any], context) -> Any:
        """根据列定义生成值"""
        col_type = column.get("type", "string")
        # 这里可以调用其他生成器
        from dataforge.core.factory import default_factory

        generator_config = GeneratorConfig(
            generator_type=col_type,
            parameters=column.get("parameters", {})
        )
        generator = default_factory.create_generator(generator_config)
        return generator.generate(context)
```

### 3. 文件路径生成器实现方案

```python
# dataforge/generators/file/filepath.py

from typing import List
from dataforge.core.generator import DataGenerator, GeneratorConfig
import random
import string

class FilePathGenerator(DataGenerator[str]):
    """文件路径生成器"""

    def __init__(self, config: GeneratorConfig):
        super().__init__(config)
        self.os_type = config.parameters.get("os", "windows")  # windows, unix, any
        self.path_type = config.parameters.get("type", "absolute")  # absolute, relative
        self.depth = config.parameters.get("depth", 3)
        self.include_spaces = config.parameters.get("include_spaces", False)
        self.include_special_chars = config.parameters.get("include_special_chars", False)

    def _generate_raw(self, context=None) -> str:
        if self.os_type == "windows" or (self.os_type == "any" and random.choice([True, False])):
            return self._generate_windows_path()
        else:
            return self._generate_unix_path()

    def _generate_windows_path(self) -> str:
        """生成Windows风格路径"""
        if self.path_type == "absolute":
            drive = random.choice(string.ascii_uppercase)
            path = f"{drive}:\\"
        else:
            path = ""

        # 生成目录层级
        for i in range(self.depth):
            dir_name = self._generate_dir_name()
            path += dir_name + "\\"

        # 生成文件名
        filename = self._generate_filename()
        path += filename

        return path

    def _generate_unix_path(self) -> str:
        """生成Unix风格路径"""
        if self.path_type == "absolute":
            path = "/"
        else:
            path = ""

        # 生成目录层级
        for i in range(self.depth):
            dir_name = self._generate_dir_name()
            path += dir_name + "/"

        # 生成文件名
        filename = self._generate_filename()
        path += filename

        return path

    def _generate_dir_name(self) -> str:
        """生成目录名"""
        length = random.randint(5, 10)
        chars = string.ascii_lowercase + string.digits
        if self.include_spaces:
            chars += " "
        if self.include_special_chars:
            chars += "_-"
        return ''.join(random.choices(chars, k=length))

    def _generate_filename(self) -> str:
        """生成文件名"""
        name_length = random.randint(5, 15)
        name = ''.join(random.choices(string.ascii_lowercase + string.digits, k=name_length))
        ext = random.choice([".txt", ".pdf", ".jpg", ".doc", ".xlsx"])
        return name + ext
```

```python
from typing import Dict, List
from dataforge.core.generator import DataGenerator, GeneratorConfig
import random

class MIMETypeGenerator(DataGenerator[str]):
    """MIME Type生成器"""

    MIME_TYPES: Dict[str, List[str]] = {
        "text": [
            "text/plain",
            "text/html",
            "text/css",
            "text/javascript",
            "text/csv",
            "text/xml"
        ],
        "image": [
            "image/jpeg",

            "image/webp",
            "image/svg+xml"
        ],
        "application": [
            "application/json",

            "application/zip",
            "application/xml",
            "application/octet-stream",
            "application/x-www-form-urlencoded"
        ],
        "audio": [
            "audio/mpeg",
            "audio/wav",
            "audio/ogg",
            "audio/aac"
        ],
        "video": [
            "video/mp4",
            "video/mpeg",
            "video/quicktime",
            "video/x-msvideo"
        ]
    }

    def __init__(self, config: GeneratorConfig):
        super().__init__(config)
        self.category = config.parameters.get("category", "any")  # text, image, application, audio, video, any

    def _generate_raw(self, context=None) -> str:
        if self.category == "any":
            # 从所有类别中随机选择
            all_types = []
            for types in self.MIME_TYPES.values():
                all_types.extend(types)
            return random.choice(all_types)
        else:
            # 从指定类别中选择
            return random.choice(self.MIME_TYPES.get(self.category, ["application/octet-stream"]))
```
